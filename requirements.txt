# nanocosmos -- pinned runtime dependencies (lockfile).
#
# Role of this file
# -----------------
# This is the *frozen* version of every package needed to reproduce the
# CUDA-13.2 / Python-3.13 development environment.  It bundles together
# the core runtime, the Cosmos backbone, the GPU (RAPIDS) extras, the
# download-script helpers, the optional visualizer, AND the dev/test
# tooling -- in other words, every ``[project.optional-dependencies]``
# group from ``pyproject.toml`` rolled into one installable manifest.
#
# When to use which file
# ----------------------
#   * ``pip install -r requirements.txt``      -- bit-for-bit reproduction
#                                                of the dev environment.
#   * ``pip install -e ".[cosmos,dev]"``       -- permissive lower bounds,
#                                                pulled from pyproject.toml.
#
# Keep ``pyproject.toml`` as the single source of truth for the package's
# logical dependency *groups*; this file just freezes specific resolved
# versions on top of that.

# Core deep learning
# torch/torchvision are the CUDA-13.2 (cu132) builds -- the driver here is
# 13.2 (nvidia-smi), and torch==2.13.0+cu132 pulls the matching CUDA 13.2
# runtime libs (cuda-toolkit==13.2.1, nvidia-cudnn-cu13==9.20.0.48,
# nvidia-nccl-cu13==2.29.7, ...), which torch pins EXACTLY -- do not bump
# cudnn/nccl independently (torch is built against those exact versions;
# a mismatch, especially NCCL, can break DDP).  torch is the single most
# serialization-sensitive library (fp8/torchao, FSDP wrapping,
# torch.save/Lightning checkpoints); leave it pinned, not resolver-drifted.
--extra-index-url https://download.pytorch.org/whl/cu132
torch==2.13.0+cu132
torchvision==0.28.0+cu132
pytorch-lightning==2.6.5
# torchmetrics is not imported directly by nanocosmos, but it is a hard
# transitive dependency of pytorch-lightning; pinned here so the frozen
# lockfile reproduces the exact installed version.
torchmetrics==1.9.0
monai==1.6.0
einops==0.8.2
# FP8 training on Blackwell+ (B200/B300): torchao float8 module-swap converts
# the DiT's Linear layers to FP8 (see nanocosmos.models.cosmos_2_5_common.fp8),
# enabled via ``model.fp8: true``.  Optional -- the code falls back to BF16 if
# torchao is missing.
# NOTE: torchao has NO cu132 wheel; the installed build is 0.17.0+cu130.  It
# runs correctly on the cu132 runtime (CUDA 13.x is minor-version compatible,
# and it targets the same torch 2.13.0 ABI) -- verified via `pip check` +
# fp8 conversion smoke test on the B300.  Re-pin to +cu132 once one ships.
torchao==0.17.0

# GPU acceleration (CUDA 13 wheels on NVIDIA's index).
# Enables:
#   - cupy-backed Mutex Watershed agglomeration (``mws_cp``) plus DLPack
#     zero-copy torch <-> cupy interop in
#     ``nanocosmos.inference.mutex_watershed``.
#   - cupy / cucim-backed EDT / Gaussian / connected components in
#     ``nanocosmos.transforms`` (optional; scipy / skimage CPU fallback).
# cupy-cuda13x works across the whole CUDA 13.x series (via cuda-pathfinder),
# so the same wheel is valid on the 13.2 runtime.
#
# NOTE: install with `pip`, not `uv pip`.  cupy-cuda13x pins
# cuda-pathfinder which uv treats as a pre-release request and refuses by
# default.  If you must use uv, pass
# `--prerelease=allow --index-strategy unsafe-best-match`.
--extra-index-url https://pypi.nvidia.com
cupy-cuda13x==14.1.1
